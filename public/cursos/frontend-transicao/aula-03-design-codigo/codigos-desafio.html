Grid responsivo + o card

auto-fill + minmax = colunas que se ajustam sozinhas.

#grid {
  display:grid;
  grid-template-columns:repeat(auto-fill, minmax(200px, 1fr));
  gap:16px;
}
.card { background:var(--card); border:1px solid var(--linha); border-radius:var(--raio); overflow:hidden; }
.card:hover { transform:translateY(-4px); box-shadow:0 16px 30px -18px rgba(74,40,120,.4); }
.capa { height:92px; }
.curtir.ativo { color:var(--rosa); border-color:var(--rosa); }

Responsivo (web → mobile)

A mágica: minmax(0, 1fr) deixa a coluna encolher sem vazar.

/* tablet: 2 colunas */
@media (max-width:620px) { #grid { grid-template-columns:repeat(2, minmax(0,1fr)); } }

/* celular: 1 coluna + esconde os links do menu */
@media (max-width:430px) { #grid { grid-template-columns:minmax(0,1fr); } }
@media (max-width:520px) { .nav-links a { display:none; } }

2 · O componente ★

Uma função que recebe 1 projeto e devolve 1 card. É isto que a gente reaproveita — a estrutura, não só o estilo. Um componente = função que devolve HTML.

function criarCard(projeto) {
  const card = document.createElement("article");
  card.className = "card";
  card.innerHTML = `
    <div class="capa" style="background:${projeto.cor}"></div>
    <div class="card-corpo">
      <span class="tag">${projeto.categoria}</span>
      <h3>${projeto.titulo}</h3>
      <button class="curtir">
        <span class="coracao">🤍</span> <span class="n">${projeto.curtidas}</span>
      </button>
    </div>`;

  // a interação do curtir vive DENTRO do componente
  const btn = card.querySelector(".curtir");
  let curtido = false, total = projeto.curtidas;
  btn.addEventListener("click", () => {
    curtido = !curtido;
    total += curtido ? 1 : -1;
    card.querySelector(".n").textContent = total;
    card.querySelector(".coracao").textContent = curtido ? "❤️" : "🤍";
    btn.classList.toggle("ativo", curtido);
  });

  return card;
}
Ponte pro React: quando você vir function Card() {} no React, é a mesma ideia — uma função que devolve um pedaço de tela. Hoje vocês fizeram no braço; o React automatiza isso.

3 · Render + filtro

Um forEach reaproveita o componente pra cada item. O filtro só muda a lista.

const grid = document.querySelector("#grid");
let filtroAtual = "Todos";

function render(lista) {
  grid.innerHTML = "";
  lista.forEach(p => grid.appendChild(criarCard(p)));
}

function aplicarFiltro() {
  const lista = (filtroAtual === "Todos")
    ? projetos
    : projetos.filter(p => p.categoria === filtroAtual);
  render(lista);
}

document.querySelectorAll(".filtro").forEach(botao => {
  botao.addEventListener("click", () => {
    document.querySelectorAll(".filtro").forEach(b => b.classList.remove("ativo"));
    botao.classList.add("ativo");
    filtroAtual = botao.dataset.cat;
    aplicarFiltro();
  });
});

📄 lib/vagas.ts 109 linhas
import { VAGAS } from "./dados";
import type { Area, FiltroVagas, Nivel, Vaga } from "./tipos";

// ============================================================================
// A camada de dados.
//
// Hoje (aula 04) tudo aqui é SÍNCRONO: os dados já estão na memória, num array.
// Ninguém precisa esperar por nada — a gente só lê, filtra e devolve.
//
// Na aula 05 essas mesmas funções vão buscar os dados de fora (uma API, um
// banco). Aí elas viram `async` e quem chama passa a usar `await`. É por isso
// que NENHUMA página conhece o `dados.ts` diretamente: todas passam por aqui.
// No dia da troca, só este arquivo muda. 💜
// ============================================================================

/** Tira acentos e deixa minúsculo — assim "junior" acha "Júnior". */
export function normalizar(texto: string) {
  return texto
    .normalize("NFD") // separa a letra do acento: "á" vira "a" + "´"
    .replace(/[̀-ͯ]/g, "") // remove os acentos que sobraram
    .toLowerCase()
    .trim();
}

/**
 * Filtra as vagas por texto, área e nível.
 * Tudo é opcional: sem filtro, devolve todas (mais recentes primeiro).
 *
 * Esta função é usada nos dois lados: no servidor (pra montar a página) e no
 * navegador (pelo filtro em tempo real). Como ela só recebe dados e devolve
 * dados, funciona igual nos dois lugares — isso se chama "função pura".
 */
export function filtrarVagas(vagas: Vaga[], filtro: FiltroVagas = {}): Vaga[] {
  const busca = normalizar(filtro.busca ?? "");

  return vagas.filter((vaga) => {
    const combinaArea = !filtro.area || vaga.area === filtro.area;
    const combinaNivel = !filtro.nivel || vaga.nivel === filtro.nivel;
    const combinaModelo = !filtro.modelo || vaga.modelo === filtro.modelo;

    const textoDaVaga = normalizar(
      [vaga.titulo, vaga.empresa, vaga.local, ...vaga.tags].join(" "),
    );
    const combinaBusca = !busca || textoDaVaga.includes(busca);

    return combinaArea && combinaNivel && combinaModelo && combinaBusca;
  });
}

/** Todas as vagas, das mais recentes pras mais antigas. */
export function listarVagas(): Vaga[] {
  return [...VAGAS].sort((a, b) => b.publicadaEm.localeCompare(a.publicadaEm));
}

/** As N vagas mais recentes — usado na home. */
export function listarVagasEmDestaque(quantidade = 3): Vaga[] {
  return listarVagas().slice(0, quantidade);
}

/**
 * Busca UMA vaga pelo slug.
 * Devolve `undefined` quando não existe — quem chama decide o que fazer
 * (na página de detalhe, a gente chama `notFound()`).
 */
export function buscarVagaPorSlug(slug: string): Vaga | undefined {
  return VAGAS.find((vaga) => vaga.slug === slug);
}

/** Todos os slugs — usado pelo `generateStaticParams` pra pré-gerar as páginas. */
export function listarSlugs(): string[] {
  return VAGAS.map((vaga) => vaga.slug);
}

/** As áreas que existem hoje, sem repetir — vira a barra de filtros. */
export function listarAreas(): Area[] {
  return [...new Set(VAGAS.map((vaga) => vaga.area))].sort();
}

/** Os níveis, na ordem de carreira (e não em ordem alfabética). */
export function listarNiveis(): Nivel[] {
  const ordem: Nivel[] = ["Estágio", "Júnior", "Pleno", "Sênior"];
  return ordem.filter((nivel) => VAGAS.some((vaga) => vaga.nivel === nivel));
}

/** Números da home: total de vagas, quantas são remotas, quantas pra quem começa. */
export function resumoDoMural() {
  return {
    total: VAGAS.length,
    remotas: VAGAS.filter((vaga) => vaga.modelo === "Remoto").length,
    paraComecar: VAGAS.filter(
      (vaga) => vaga.nivel === "Estágio" || vaga.nivel === "Júnior",
    ).length,
    empresas: new Set(VAGAS.map((vaga) => vaga.empresa)).size,
  };
}

/** "publicada há 3 dias" — a partir da data ISO da vaga. */
export function formatarPublicacao(dataISO: string): string {
  const publicada = new Date(`${dataISO}T12:00:00`);
  const hoje = new Date();
  const dias = Math.round(
    (hoje.getTime() - publicada.getTime()) / (1000 * 60 * 60 * 24),
  );

  if (dias <= 0) return "publicada hoje";
  if (dias === 1) return "publicada ontem";
  if (dias < 30) return `publicada há ${dias} dias`;
  return `publicada em ${publicada.toLocaleDateString("pt-BR")}`;
}

components/MuralDeVagas.tsx

⭐ A fronteira do projeto. É aqui que mora o useState da busca e dos filtros. Leia com atenção o comentário sobre estado derivado: a lista filtrada é calculada, nunca guardada — é o erro nº 1 de quem começa.

📄 components/MuralDeVagas.tsx 80 linhas
"use client";

import { useState } from "react";
import { filtrarVagas } from "@/lib/vagas";
import type { Area, Nivel, Vaga } from "@/lib/tipos";
import Filtros from "./Filtros";
import VagaCard from "./VagaCard";
import estilos from "./MuralDeVagas.module.css";

// ============================================================================
// O mural com busca e filtros — tudo acontecendo NO NAVEGADOR.
//
// A página manda a lista completa de vagas por prop. Aqui a gente guarda o que
// a pessoa digitou/escolheu e recalcula o que aparece na tela. Sem recarregar,
// sem esperar servidor: você digita e a lista muda na mesma hora.
// ============================================================================

type Props = { vagas: Vaga[]; areas: Area[]; niveis: Nivel[] };

export default function MuralDeVagas({ vagas, areas, niveis }: Props) {
  // Um estado por "coisa que a pessoa controla". Só isso.
  const [busca, setBusca] = useState("");
  const [area, setArea] = useState("");
  const [nivel, setNivel] = useState("");

  // ⚠️ ESTADO DERIVADO — leia com atenção, é o erro nº 1 de quem começa.
  //
  // A lista filtrada NÃO é um useState. Ela é CALCULADA a partir dos outros
  // estados, a cada renderização. Se você criasse um `useState(vagasFiltradas)`
  // teria que lembrar de atualizar ele em todo lugar que mexe num filtro — e
  // um dia esqueceria. Regra: se dá pra calcular, não guarde em estado.
  const visiveis = filtrarVagas(vagas, { busca, area, nivel });

  const temFiltro = Boolean(busca || area || nivel);

  function limpar() {
    setBusca("");
    setArea("");
    setNivel("");
  }

  return (
    <>
      <Filtros
        areas={areas}
        niveis={niveis}
        busca={busca}
        area={area}
        nivel={nivel}
        aoMudarBusca={setBusca}
        aoMudarArea={setArea}
        aoMudarNivel={setNivel}
        aoLimpar={temFiltro ? limpar : undefined}
      />

      <p className={estilos.contador} aria-live="polite">
        {visiveis.length === vagas.length
          ? `${vagas.length} vagas abertas`
          : `${visiveis.length} de ${vagas.length} vagas`}
      </p>

      {visiveis.length === 0 ? (
        <div className="vazio">
          <strong>Nenhuma vaga com esses filtros 😔</strong>
          Tente uma busca mais curta ou{" "}
          <button type="button" onClick={limpar} className={estilos.linkBotao}>
            limpe os filtros
          </button>
          .
        </div>
      ) : (
        <div className={estilos.grade}>
          {visiveis.map((vaga) => (
            <VagaCard key={vaga.slug} vaga={vaga} />
          ))}
        </div>
      )}
    </>
  );
}

"use client";

import { useEffect, useState } from "react";
import estilos from "./SalvarButton.module.css";

// ============================================================================
// "use client" na primeira linha = este componente também roda NO NAVEGADOR.
//
// Precisamos disso porque ele tem duas coisas que só existem no navegador:
//   • estado que muda (useState)
//   • um evento de clique (onClick)
//
// Regra de bolso: deixe "use client" o mais FUNDO possível na árvore.
// Aqui, só o botãozinho é cliente — o card inteiro continua sendo servidor.
// ============================================================================

const CHAVE = "leque-de-vagas:salvas";

function lerSalvas(): string[] {
  try {
    return JSON.parse(localStorage.getItem(CHAVE) ?? "[]");
  } catch {
    return [];
  }
}

export default function SalvarButton({
  slug,
  titulo,
}: {
  slug: string;
  titulo: string;
}) {
  const [salva, setSalva] = useState(false);

  // O servidor não tem `localStorage`. Se a gente lesse ele direto no corpo do
  // componente, o HTML do servidor sairia diferente do que o navegador desenha
  // e o React reclamaria ("hydration mismatch"). Por isso lemos no useEffect,
  // que só roda depois que o componente já apareceu na tela.
  useEffect(() => {
    setSalva(lerSalvas().includes(slug));
  }, [slug]);

  function alternar() {
    const salvas = lerSalvas();
    const novas = salva
      ? salvas.filter((s) => s !== slug)
      : [...salvas, slug];

    localStorage.setItem(CHAVE, JSON.stringify(novas));
    setSalva(!salva);
  }

  return (
    <button
      type="button"
      onClick={alternar}
      className={`${estilos.botao} ${salva ? estilos.ativo : ""}`}
      aria-pressed={salva}
      aria-label={salva ? `Remover ${titulo} das salvas` : `Salvar ${titulo}`}
      title={salva ? "Remover das salvas" : "Salvar vaga"}
    >
      {salva ? "★" : "☆"}
    </button>
  );
}

// O formulário de login.

// ⚠️ Ele NÃO faz login de verdade — não existe servidor, senha ou conta ainda.
// Hoje o assunto é FRONTEND: input controlado, validação, mensagem de erro e

/** O formato dos erros: uma mensagem por campo (ou nenhuma). */
type Erros = { email?: string; senha?: string };

export default function FormEntrar() {
  // Um estado por campo. É o que "input controlado" significa: o valor que
  // aparece na tela vem do estado, e toda tecla digitada atualiza o estado.
  const [email, setEmail] = useState("");
  const [senha, setSenha] = useState("");

  const [erros, setErros] = useState<Erros>({});
  const [verSenha, setVerSenha] = useState(false);
  const [enviando, setEnviando] = useState(false);
  const [enviado, setEnviado] = useState(false);

  /** Devolve um objeto com os erros encontrados. Vazio = está tudo certo. */
  function validar(): Erros {
    const novos: Erros = {};

    if (!email.trim()) {
      novos.email = "Escreva seu e-mail.";
    } else if (!email.includes("@") || !email.includes(".")) {
      novos.email = "Esse e-mail não parece completo.";
    }

    if (!senha) {
      novos.senha = "Escreva sua senha.";
    } else if (senha.length < 6) {
      novos.senha = "A senha tem pelo menos 6 caracteres.";
    }

    return novos;
  }

  function enviar(evento: React.FormEvent) {
    // Sem isto o navegador recarrega a página inteira ao enviar o formulário.
    evento.preventDefault();

    const novos = validar();
    setErros(novos);

    // Object.keys transforma o objeto numa lista de chaves.
    // Lista vazia = nenhum erro = pode seguir.
    if (Object.keys(novos).length > 0) return;

    setEnviando(true);

    // Fingindo a demora de um servidor, só pra você ver o estado de carregando
    // funcionando. Na aula 05, aqui vai entrar a chamada de verdade.
    setTimeout(() => {
      setEnviando(false);
      setEnviado(true);
    }, 1200);
  }

  if (enviado) {
    return (
      <div className={estilos.sucesso}>
        <span>✅</span>
        <div>
          <strong>Formulário validado!</strong>
          Os dados de <b>{email}</b> passaram por toda a validação. Na{" "}
          <b>aula 05</b> este botão vai falar com o servidor e te deixar entrar
          de verdade. 💜
        </div>
      </div>
    );
  }

    <form className={estilos.form} onSubmit={enviar} noValidate>
      <CampoTexto
        id="email"
        rotulo="E-mail"
        tipo="email"
        valor={email}
        // Ao digitar, o erro daquele campo some: corrigir tem que dar
        // sensação de progresso, não de castigo.
        aoMudar={(valor) => {
          setEmail(valor);
          setErros((atuais) => ({ ...atuais, email: undefined }));
        }}
        erro={erros.email}
        placeholder="voce@email.com"
        autoComplete="email"
      />

      <CampoTexto
        id="senha"
        rotulo="Senha"
        tipo={verSenha ? "text" : "password"}
        valor={senha}
        aoMudar={(valor) => {
          setSenha(valor);
          setErros((atuais) => ({ ...atuais, senha: undefined }));
        }}
        erro={erros.senha}
        placeholder="••••••"
        autoComplete="current-password"
        acessorio={
          <button
            type="button"
            className={estilos.olho}
            onClick={() => setVerSenha(!verSenha)}
            aria-label={verSenha ? "Ocultar senha" : "Mostrar senha"}
          >
            {verSenha ? "🙈" : "👁️"}

      </button>

      </p>

function medirForca(senha: string): number {
  let pontos = 0;
  if (senha.length >= 8) pontos++;
  if (/[A-Z]/.test(senha) && /[a-z]/.test(senha)) pontos++;
  if (/[0-9]/.test(senha) || /[^A-Za-z0-9]/.test(senha)) pontos++;
  return pontos;
}

// Metadata dinâmica: cada vaga tem seu próprio <title> e descrição.
// É isso que faz o link ficar bonito quando alguém compartilha no WhatsApp.
export async function generateMetadata({ params }: Props): Promise<Metadata> {
  const { slug } = await params;
  const vaga = buscarVagaPorSlug(slug);

  if (!vaga) return { title: "Vaga não encontrada" };

  return {
    title: `${vaga.titulo} — ${vaga.empresa}`,
    description: vaga.resumo,
    openGraph: {
      title: `${vaga.titulo} · ${vaga.empresa}`,
      description: vaga.resumo,
      type: "article",
    },
  };
}

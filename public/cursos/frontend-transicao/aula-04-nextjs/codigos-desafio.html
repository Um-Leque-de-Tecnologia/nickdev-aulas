lib/vagas.ts

A camada de dados. Todas as funções são async de propósito: hoje elas leem um array, na aula 05 vão ler um banco de verdade. Como as páginas já usam await, no dia da troca nenhuma página muda.

📄 lib/vagas.ts 106 linhas
import { VAGAS } from "./dados";
import type { Area, FiltroVagas, Nivel, Vaga } from "./tipos";

// ============================================================================
// A camada de dados.
//
// Todas as funções são `async` de propósito: hoje elas leem um array, mas
// amanhã vão ler um banco de dados de verdade. Como as páginas já usam
// `await`, no dia da troca NENHUMA página precisa mudar. 💜
// ============================================================================

/** Simula a demora de uma consulta ao banco, pra a gente ver o `loading.tsx`. */
function demora(ms: number) {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

/** Tira acentos e deixa minúsculo — assim "junior" acha "Júnior". */
function normalizar(texto: string) {
  return texto
    .normalize("NFD") // separa a letra do acento: "á" vira "a" + "´"
    .replace(/[\u0300-\u036f]/g, "") // remove os acentos que sobraram
    .toLowerCase()
    .trim();
}

/**
 * Lista as vagas, aplicando busca por texto e filtros de área/nível.
 * Tudo é opcional: sem filtro, devolve todas (mais recentes primeiro).
 */
export async function listarVagas(filtro: FiltroVagas = {}): Promise<Vaga[]> {
  await demora(400);

  const busca = normalizar(filtro.busca ?? "");

  return VAGAS.filter((vaga) => {
    const combinaArea = !filtro.area || vaga.area === filtro.area;
    const combinaNivel = !filtro.nivel || vaga.nivel === filtro.nivel;

    const textoDaVaga = normalizar(
      [vaga.titulo, vaga.empresa, vaga.local, ...vaga.tags].join(" "),
    );
    const combinaBusca = !busca || textoDaVaga.includes(busca);

    return combinaArea && combinaNivel && combinaBusca;
  }).sort((a, b) => b.publicadaEm.localeCompare(a.publicadaEm));
}

/** As N vagas mais recentes — usado na home. */
export async function listarVagasEmDestaque(quantidade = 3): Promise<Vaga[]> {
  const vagas = await listarVagas();
  return vagas.slice(0, quantidade);
}

/**
 * Busca UMA vaga pelo slug.
 * Devolve `undefined` quando não existe — quem chama decide o que fazer
 * (na página de detalhe, a gente chama `notFound()`).
 */
export async function buscarVagaPorSlug(
  slug: string,
): Promise<Vaga | undefined> {
  await demora(200);
  return VAGAS.find((vaga) => vaga.slug === slug);
}

/** Todos os slugs — usado pelo `generateStaticParams` pra pré-gerar as páginas. */
export async function listarSlugs(): Promise<string[]> {
  return VAGAS.map((vaga) => vaga.slug);
}

/** As áreas que existem hoje, sem repetir — vira a barra de filtros. */
export async function listarAreas(): Promise<Area[]> {
  return [...new Set(VAGAS.map((vaga) => vaga.area))].sort();
}

/** Os níveis, na ordem de carreira (e não em ordem alfabética). */
export async function listarNiveis(): Promise<Nivel[]> {
  const ordem: Nivel[] = ["Estágio", "Júnior", "Pleno", "Sênior"];
  return ordem.filter((nivel) => VAGAS.some((vaga) => vaga.nivel === nivel));
}

/** Números da home: total de vagas, quantas são remotas, quantas pra quem começa. */
export async function resumoDoMural() {
  return {
    total: VAGAS.length,
    remotas: VAGAS.filter((vaga) => vaga.modelo === "Remoto").length,
    paraComecar: VAGAS.filter(
      (vaga) => vaga.nivel === "Estágio" || vaga.nivel === "Júnior",
    ).length,
    empresas: new Set(VAGAS.map((vaga) => vaga.empresa)).size,
  };
}

/** "publicada há 3 dias" — a partir da data ISO da vaga. */
export function formatarPublicacao(dataISO: string): string {
  const publicada = new Date(`${dataISO}T12:00:00`);
  const hoje = new Date();
  const dias = Math.round(
    (hoje.getTime() - publicada.getTime()) / (1000 * 60 * 60 * 24),
  );

  if (dias <= 0) return "publicada hoje";
  if (dias === 1) return "publicada ontem";
  if (dias < 30) return `publicada há ${dias} dias`;
  return `publicada em ${publicada.toLocaleDateString("pt-BR")}`;
}

components/SalvarButton.tsx

⭐ O único arquivo do projeto com "use client". Ele precisa disso por dois motivos: tem estado (useState) e tem evento (onClick). Repare por que o localStorage é lido dentro do useEffect — o comentário no código explica.

📄 components/SalvarButton.tsx 66 linhas
"use client";

import { useEffect, useState } from "react";
import estilos from "./SalvarButton.module.css";

// ============================================================================
// "use client" na primeira linha = este componente também roda NO NAVEGADOR.
//
// Precisamos disso porque ele tem duas coisas que só existem no navegador:
//   • estado que muda (useState)
//   • um evento de clique (onClick)
//
// Regra de bolso: deixe "use client" o mais FUNDO possível na árvore.
// Aqui, só o botãozinho é cliente — o card inteiro continua sendo servidor.
// ============================================================================

const CHAVE = "leque-de-vagas:salvas";

function lerSalvas(): string[] {
  try {
    return JSON.parse(localStorage.getItem(CHAVE) ?? "[]");
  } catch {
    return [];
  }
}

export default function SalvarButton({
  slug,
  titulo,
}: {
  slug: string;
  titulo: string;
}) {
  const [salva, setSalva] = useState(false);

  // O servidor não tem `localStorage`. Se a gente lesse ele direto no corpo do
  // componente, o HTML do servidor sairia diferente do que o navegador desenha
  // e o React reclamaria ("hydration mismatch"). Por isso lemos no useEffect,
  // que só roda depois que o componente já apareceu na tela.
  useEffect(() => {
    setSalva(lerSalvas().includes(slug));
  }, [slug]);

  function alternar() {
    const salvas = lerSalvas();
    const novas = salva
      ? salvas.filter((s) => s !== slug)
      : [...salvas, slug];

    localStorage.setItem(CHAVE, JSON.stringify(novas));
    setSalva(!salva);
  }

  return (
    <button
      type="button"
      onClick={alternar}
      className={`${estilos.botao} ${salva ? estilos.ativo : ""}`}
      aria-pressed={salva}
      aria-label={salva ? `Remover ${titulo} das salvas` : `Salvar ${titulo}`}
      title={salva ? "Remover das salvas" : "Salvar vaga"}
    >
      {salva ? "★" : "☆"}
    </button>
  );
}

components/Filtros.tsx

Interativo e Componente de Servidor. Como? A busca é um <form> com method GET e os filtros são <Link> que trocam a query string. Funciona até com o JavaScript desligado.

📄 components/Filtros.tsx 118 linhas
import Link from "next/link";
import type { Area, Nivel } from "@/lib/tipos";
import estilos from "./Filtros.module.css";

// Este componente é de SERVIDOR e mesmo assim é interativo. Como?
//   • a busca é um <form> com method GET → o próprio navegador monta a URL
//     /vagas?busca=react e o Next re-renderiza a página no servidor
//   • os filtros são <Link> que trocam a query string
//
// Ou seja: filtro e busca funcionam ATÉ COM O JAVASCRIPT DESLIGADO, e o
// estado da tela fica guardado na URL (dá pra copiar e mandar pra alguém).

type Props = {
  areas: Area[];
  niveis: Nivel[];
  busca: string;
  areaAtiva: string;
  nivelAtivo: string;
};

/** Monta uma URL /vagas mantendo os filtros atuais e trocando só um deles. */
function montarUrl(
  atual: { busca: string; area: string; nivel: string },
  troca: Partial<{ area: string; nivel: string }>,
) {
  const params = new URLSearchParams();
  const alvo = { ...atual, ...troca };

  if (alvo.busca) params.set("busca", alvo.busca);
  if (alvo.area) params.set("area", alvo.area);
  if (alvo.nivel) params.set("nivel", alvo.nivel);

  const query = params.toString();
  return query ? `/vagas?${query}` : "/vagas";
}

export default function Filtros({
  areas,
  niveis,
  busca,
  areaAtiva,
  nivelAtivo,
}: Props) {
  const atual = { busca, area: areaAtiva, nivel: nivelAtivo };
  const temFiltro = Boolean(busca || areaAtiva || nivelAtivo);

  return (
    <section className={estilos.painel} aria-label="Buscar e filtrar vagas">
      <form className={estilos.busca} action="/vagas">
        <input
          type="search"
          name="busca"
          defaultValue={busca}
          placeholder="Busque por cargo, empresa, tecnologia…"
          aria-label="Buscar vagas"
        />
        {/* mantemos os filtros ativos ao enviar a busca */}
        {areaAtiva && <input type="hidden" name="area" value={areaAtiva} />}
        {nivelAtivo && <input type="hidden" name="nivel" value={nivelAtivo} />}
        <button type="submit" className="btn btn-primario">
          Buscar
        </button>
      </form>

      <div className={estilos.grupo}>
        <span className={estilos.rotulo}>Área</span>
        <div className={estilos.chips}>
          <Link
            href={montarUrl(atual, { area: "" })}
            className={`${estilos.chip} ${!areaAtiva ? estilos.chipAtivo : ""}`}
          >
            Todas
          </Link>
          {areas.map((area) => (
            <Link
              key={area}
              href={montarUrl(atual, { area })}
              className={`${estilos.chip} ${
                areaAtiva === area ? estilos.chipAtivo : ""
              }`}
            >
              {area}
            </Link>
          ))}
        </div>
      </div>

      <div className={estilos.grupo}>
        <span className={estilos.rotulo}>Nível</span>
        <div className={estilos.chips}>
          <Link
            href={montarUrl(atual, { nivel: "" })}
            className={`${estilos.chip} ${!nivelAtivo ? estilos.chipAtivo : ""}`}
          >
            Todos
          </Link>
          {niveis.map((nivel) => (
            <Link
              key={nivel}
              href={montarUrl(atual, { nivel })}
              className={`${estilos.chip} ${
                nivelAtivo === nivel ? estilos.chipAtivo : ""
              }`}
            >
              {nivel}
            </Link>
          ))}
        </div>
      </div>

      {temFiltro && (
        <Link href="/vagas" className={estilos.limpar}>
          ✕ limpar filtros
        </Link>
      )}
    </section>
  );
}
